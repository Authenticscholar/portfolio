def load_tea_ordered() -> tuple[list[tuple[str, float, float]], dict[str, tuple[float, float]]]:
    """Return capture technologies in sheet order (name, low, high) and shared param bounds."""
    df = pd.read_excel(WORKBOOK, sheet_name="21_TEA_Parameters", engine="openpyxl")
    df = df.rename(columns=str.lower)
    capture: list[tuple[str, float, float]] = []
    shared: dict[str, tuple[float, float]] = {}
    for _, row in df.iterrows():
        name = str(row["parameter"]).strip()
        lo, hi = float(row["low"]), float(row["high"])
        lname = name.lower()
        if (
            "capture" in lname
            or "liquefaction" in lname
            or ("calcium" in lname and "looping" in lname)
        ):
            capture.append((name, lo, hi))
        elif "pipeline transport" in lname:
            shared["transport"] = (lo, hi)
        elif "offshore storage" in lname or ("storage" in lname and "offshore" in lname):
            shared["storage"] = (lo, hi)
        elif "energy penalty" in lname:
            shared["energy_penalty"] = (lo, hi)
        elif "grid emission" in lname:
            shared["grid_ef"] = (lo, hi)
    return capture, shared


def tech_display_name(full: str) -> str:

# R mirror: Monte Carlo LCO₂A (USD/t) for Post-combustion MEA — run after:
#   install.packages("triangle")
# Rscript monte_carlo_mirror.R

set.seed(42)
N <- 10000L

if (!requireNamespace("triangle", quietly = TRUE)) {
  stop("Install triangle: install.packages('triangle')")
}

tri_draw <- function(lo, hi) {
  mode <- (lo + hi) / 2
  triangle::rtriangle(N, a = lo, b = hi, c = mode)
}

c_cap <- tri_draw(79, 125)
c_tr <- tri_draw(4.6, 11.9)
c_st <- tri_draw(5.3, 15.8)
eta <- tri_draw(2.5, 4.5)
grid <- tri_draw(0.05, 0.12)
indirect <- eta * (grid / 3.6)
lco2a <- (c_cap + c_tr + c_st) / (1 - indirect)

cat(sprintf("MEA median USD/t: %.4f (Python ref ~131.3767)\n", stats::median(lco2a)))
